  let currentTopic = 'all';
  let currentStatus = 'all';
  let currentSearch = '';
  let currentView = 'grid';

  function filterKits() {
    return KITS_DATA.filter(kit => {
      if (currentTopic !== 'all' && kit.topic !== currentTopic) return false;
      if (currentStatus !== 'all' && kit.status !== currentStatus) return false;
      if (currentSearch.trim() !== '') {
        const q = currentSearch.toLowerCase();
        const matchTitle = kit.title.toLowerCase().includes(q);
        const matchDesc = kit.desc.toLowerCase().includes(q);
        const matchCat = kit.topicLabel.toLowerCase().includes(q);
        const matchDeliv = kit.delivs.some(d => d.toLowerCase().includes(q));
        if (!matchTitle && !matchDesc && !matchCat && !matchDeliv) return false;
      }
      return true;
    });
  }

  function renderContextualSuite() {
    const showcaseEl = document.getElementById('bundles-showcase');
    const ctxBannerEl = document.getElementById('suite-contextual-banner');

    if (currentTopic === 'all') {
      showcaseEl.style.display = 'flex';
      ctxBannerEl.style.display = 'none';
    } else {
      showcaseEl.style.display = 'none';
      const suite = SUITES_CONFIG[currentTopic];
      if (suite) {
        ctxBannerEl.style.display = 'block';
        document.getElementById('suite-ctx-icon').textContent = suite.icon;
        document.getElementById('suite-ctx-title').textContent = suite.title;
        document.getElementById('suite-ctx-tag').textContent = suite.tag;
        document.getElementById('suite-ctx-desc').textContent = suite.desc;
        document.getElementById('suite-ctx-price').textContent = suite.price;
        document.getElementById('suite-ctx-orig').textContent = suite.origPrice ? `${suite.origPrice} individually` : '';

        const btnBox = document.getElementById('suite-ctx-btn-box');
        if (suite.available) {
          btnBox.innerHTML = `
            <a href="${suite.checkoutUrl}" class="btn-suite-buy-full lemonsqueezy-button">
              Download Complete Suite • ${suite.price} →
            </a>
          `;
        } else {
          btnBox.innerHTML = `
            <span class="btn-kit-disabled">Coming soon</span>
          `;
        }
      } else {
        ctxBannerEl.style.display = 'none';
      }
    }
  }

  function setTopic(topic) {
    currentTopic = topic;
    document.querySelectorAll('#topic-pills-container .topic-pill').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.topic === topic);
    });
    render();
  }

  function setStatus(status) {
    currentStatus = status;
    document.querySelectorAll('.status-segment .btn-segment').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.status === status);
    });
    render();
  }

  function handleSearch() {
    currentSearch = document.getElementById('search-input').value;
    render();
  }

  function setView(view) {
    currentView = view;
    document.getElementById('btn-view-grid').classList.toggle('active', view === 'grid');
    document.getElementById('btn-view-table').classList.toggle('active', view === 'table');
    render();
  }

  function openModal(kitId) {
    const kit = KITS_DATA.find(k => k.id === kitId);
    if (!kit) return;

    document.getElementById('modal-icon').textContent = kit.icon;
    document.getElementById('modal-title').textContent = kit.title;
    document.getElementById('modal-cat').textContent = `${kit.topicLabel} • ${kit.statusLabel}`;
    document.getElementById('modal-desc').textContent = kit.desc;
    document.getElementById('modal-price').textContent = `${kit.price} (was ${kit.origPrice})`;

    const delivsEl = document.getElementById('modal-delivs');
    delivsEl.innerHTML = kit.delivs.map(d => `<span class="deliv-chip" style="font-size: 12px; padding: 4px 8px;">${d}</span>`).join('');

    const featuresEl = document.getElementById('modal-features');
    featuresEl.innerHTML = kit.features.map(f => `
      <li>
        <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><polyline points="20 6 9 17 4 12"/></svg>
        <span>${f}</span>
      </li>
    `).join('');

    const buyBtn = document.getElementById('modal-buy-link');
    if (kit.status === 'available') {
      buyBtn.href = kit.checkoutUrl;
      buyBtn.textContent = `Download Kit • ${kit.price}`;
      buyBtn.style.display = 'inline-flex';
    } else {
      buyBtn.style.display = 'none';
    }

    const guideBtn = document.getElementById('modal-guide-link');
    if (kit.guideUrl) {
      guideBtn.href = kit.guideUrl;
      guideBtn.innerHTML = 'Leer Guía Gratuita 📖';
      guideBtn.style.display = 'inline-flex';
      guideBtn.style.opacity = '1';
      guideBtn.style.cursor = 'pointer';
      guideBtn.style.borderStyle = 'solid';
    } else {
      guideBtn.removeAttribute('href');
      guideBtn.innerHTML = 'Guía Próximamente ⏳';
      guideBtn.style.display = 'inline-flex';
      guideBtn.style.opacity = '0.55';
      guideBtn.style.cursor = 'default';
      guideBtn.style.borderStyle = 'dashed';
    }

    document.getElementById('modal-backdrop').style.display = 'flex';
    document.body.style.overflow = 'hidden';
  }

  function closeModal() {
    document.getElementById('modal-backdrop').style.display = 'none';
    document.body.style.overflow = 'auto';
  }

  function closeModalOnBackdrop(e) {
    if (e.target.id === 'modal-backdrop') closeModal();
  }

  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') closeModal();
  });

  // Initial render
  render();

  // Smooth scroll and topic filtering for URL hashes (#suite-01, #pack-dafo, etc.)
  if (window.location.hash) {
    const h = window.location.hash.toLowerCase();
    if (h === '#suite-01' || h === '#suite-1' || h === '#estrategia') setTopic('estrategia');
    else if (h === '#suite-02' || h === '#suite-2' || h === '#decisiones') setTopic('decisiones');
    else if (h === '#suite-03' || h === '#suite-3' || h === '#operaciones') setTopic('operaciones');
    else if (h === '#suite-sdd' || h === '#software') setTopic('software');

    setTimeout(() => {
      const target = document.querySelector(window.location.hash);
      if (target) {
        target.scrollIntoView({ behavior: 'smooth', block: 'center' });
        target.classList.add('highlight-pulse');
        setTimeout(() => target.classList.remove('highlight-pulse'), 2500);
      }
    }, 350);
  }
