  function openModal(kitId) {
    const kit = KITS_DATA.find(k => k.id === kitId) || Object.values(SUITES_CONFIG).find(s => s.id === kitId);
    if (!kit) return;

    document.getElementById('modal-icon').textContent = kit.icon;
    document.getElementById('modal-title').textContent = kit.title;
    document.getElementById('modal-cat').textContent = kit.topicLabel;
    document.getElementById('modal-desc').textContent = kit.desc;
    document.getElementById('modal-price').textContent = kit.origPrice ? `${kit.price} (was ${kit.origPrice})` : kit.price;

    renderModalPreview(kit, false);

    const delivsEl = document.getElementById('modal-delivs');
    delivsEl.innerHTML = kit.delivs.map(d => `<span class="deliv-chip" style="font-size: 12px; padding: 4px 8px;">${d}</span>`).join('');

    const featuresEl = document.getElementById('modal-features');
    featuresEl.innerHTML = kit.features.map(f => `
      <li>
        <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><polyline points="20 6 9 17 4 12"/></svg>
        <span>${f}</span>
      </li>
    `).join('');

    const buyBtn = document.getElementById('modal-buy-link');
    if (kit.status === 'available') {
      buyBtn.href = kit.checkoutUrl;
      buyBtn.className = 'btn-kit-buy lemonsqueezy-button';
      buyBtn.textContent = kit.id.startsWith('suite') ? `Download Suite • ${kit.price}` : `Download Pack • ${kit.price}`;
      buyBtn.style.display = 'inline-flex';
      buyBtn.onclick = null;
    } else {
      buyBtn.href = '#';
      buyBtn.className = 'btn-kit-waitlist';
      buyBtn.textContent = 'Join Waitlist (19€ Early-Bird) 🧬';
      buyBtn.style.display = 'inline-flex';
      buyBtn.onclick = function(e) { e.preventDefault(); closeModal(); openWaitlistModal(kit.id); };
    }

    const guideBtn = document.getElementById('modal-guide-link');
    if (kit.guideUrl) {
      guideBtn.href = kit.guideUrl;
      guideBtn.innerHTML = 'Technical Article 📖';
      guideBtn.style.display = 'inline-flex';
      guideBtn.style.opacity = '1';
      guideBtn.style.cursor = 'pointer';
      guideBtn.style.borderStyle = 'solid';
      guideBtn.onclick = null;
    } else {
      guideBtn.href = '#';
      guideBtn.innerHTML = 'Technical article coming soon';
      guideBtn.style.display = 'inline-flex';
      guideBtn.style.opacity = '0.65';
      guideBtn.style.cursor = 'not-allowed';
      guideBtn.style.borderStyle = 'dashed';
      guideBtn.onclick = (e) => e.preventDefault();
    }

    document.getElementById('modal-backdrop').style.display = 'flex';
    document.body.style.overflow = 'hidden';

    trackKitEvent('view_kit_details', {
      item_id: kitId,
      item_name: kit.title,
      language: 'en'
    });

    if (window.createLemonSqueezy) window.createLemonSqueezy();
  }

    const isES = document.documentElement.lang !== 'en';

    if (feedback) feedback.style.display = 'none';
    const form = document.getElementById('waitlist-form-catalog');
    if (form) form.reset();
    backdrop.dataset.kitId = kitId || 'pack-sdd';
    backdrop.style.display = 'flex';
    document.body.style.overflow = 'hidden';
  }

    const backdrop = document.getElementById('waitlist-modal-backdrop');
    if (backdrop) backdrop.style.display = 'none';
    document.body.style.overflow = 'auto';
  }

  }

  async function submitCatalogWaitlist(e) {
    e.preventDefault();
    const isES = document.documentElement.lang !== 'en';
    const lang = isES ? 'es' : 'en';
    const name = document.getElementById('wl-name').value.trim();
    const email = document.getElementById('wl-email').value.trim();
    const btn = document.getElementById('wl-submit-btn');
    const feedback = document.getElementById('wl-feedback');
    const origText = btn.textContent;
    const kitId = document.getElementById('waitlist-modal-backdrop')?.dataset.kitId || 'pack-sdd';
    const isLocalhost = window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1';

    btn.disabled = true;
    btn.textContent = isES ? 'Registrando...' : 'Registering...';
    feedback.style.display = 'none';

    function handleSuccess(devMsg) {
      feedback.style.display = 'block';
      feedback.style.color = '#34d399';
      feedback.textContent = devMsg || (isES
        ? '✅ ¡Plaza reservada con éxito! Te avisaremos con acceso prioritario y precio Early-Bird de 19€.'
        : '✅ Spot reserved successfully! We will notify you with priority access and 19€ Early-Bird pricing.');
      document.getElementById('waitlist-form-catalog').reset();

      trackKitEvent('join_waitlist_helix', {
        item_id: kitId,
        item_name: 'Datalaria Helix',
        language: lang,
        source: 'catalog_modal'
      });

      setTimeout(() => {
        closeWaitlistModal();
      }, 2500);
    }

    try {
      const res = await fetch('/.netlify/functions/subscribe', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name, email, language: lang, list: 'helix' })
      });

      if (isLocalhost && (!res.ok && res.status === 404)) {
        console.info('[Catalog Waitlist Dev] Registro simulado en localhost (Hugo Server):', {
          name, email, language: lang, list: 'helix', brevoListId: isES ? 5 : 6
        });
        handleSuccess(isES
          ? '✅ ¡Plaza reservada! [Modo Localhost: En producción Netlify registrará el contacto en Brevo (Lista #' + (isES ? '5' : '6') + ')].'
          : '✅ Spot reserved! [Localhost Mode: In Netlify production contact is registered in Brevo (List #' + (isES ? '5' : '6') + ')].');
        return;
      }

      if (!res.ok) throw new Error('HTTP ' + res.status);

      handleSuccess();
    } catch (err) {
      if (isLocalhost) {
        console.info('[Catalog Waitlist Dev] Fallback localhost activado:', err);
        handleSuccess(isES
          ? '✅ ¡Plaza reservada! [Modo Localhost: En producción Netlify registrará el contacto en Brevo (Lista #' + (isES ? '5' : '6') + ')].'
          : '✅ Spot reserved! [Localhost Mode: In Netlify production contact is registered in Brevo (List #' + (isES ? '5' : '6') + ')].');
      } else {
        feedback.style.display = 'block';
        feedback.style.color = '#f87171';
        feedback.textContent = isES
          ? '⚠️ Ocurrió un error al registrarte. Inténtalo de nuevo en unos segundos.'
          : '⚠️ An error occurred. Please try again in a few seconds.';
      }
    } finally {
      btn.disabled = false;
      btn.textContent = origText;
    }
  }
